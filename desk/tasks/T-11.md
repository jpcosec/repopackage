### `src/repopackage/cli/eval/commands.py`

```python
@eval_app.command(name="run")
def run_eval(
    last_phase: bool = typer.Option(False, "--last-phase", help="Eval last active phase"),
    phase_id: Optional[str] = typer.Option(None, "--phase", help="Specific phase ID"),
):
    ws = Workspace()
    if last_phase:
        phase = ws.last_phase()
    elif phase_id:
        phase = ws.phase(phase_id)
    else:
        phase = ws.last_phase()  # default
    
    result = phase.eval()
    # Rich output: passed/failed counts + output
    if result.success:
        typer.echo(f"✓ Phase {phase.id}: {result.passed} passed")
    else:
        typer.echo(f"✗ Phase {phase.id}: {result.failed} failed")
        typer.echo(result.output)
    raise typer.Exit(0 if result.success else 1)
```
